# Physics and medical imaging stack
# Confirmed working on macOS Apple Silicon (M-series), Python 3.11
#
# INSTALL IN THIS EXACT ORDER:
#
# Step 1 — create and activate conda environment:
#   conda create -n adaptive-rt python=3.11 -c conda-forge
#   conda activate adaptive-rt
#
# Step 2 — install scientific stack via conda (handles binaries cleanly):
#   conda install -c conda-forge "numpy>=2.0,<2.3" scipy=1.13.1 pandas=2.2.3 matplotlib=3.9.4 -y
#
# Step 3 — install pyRadPlan without its AI dependency chain:
#   pip install pyRadPlan==0.4.0 --no-deps
#
# Step 4 — install pyRadPlan's actual runtime dependencies:
#   pip install -r requirements-physics.txt
#
# Step 5 — install pydantic-ai (satisfies pyRadPlan import check, no deps):
#   pip install "pydantic-ai>=1.60,<2.0" --no-deps
#
# WHY --no-deps FOR pyRadPlan AND pydantic-ai:
#   pyRadPlan depends on pydantic-ai which pulls in Anthropic, Cohere,
#   MistralAI, HuggingFace SDKs that conflict with each other.
#   We only need pyRadPlan's physics engine (calc_dose_influence,
#   generate_stf) — none of those AI SDKs are used in this project.
